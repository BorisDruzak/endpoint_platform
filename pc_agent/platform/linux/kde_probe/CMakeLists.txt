cmake_minimum_required(VERSION 3.20)
project(EndpointKdeProbe LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui DBus)
find_package(KF6IdleTime 6.0 REQUIRED)
add_executable(endpoint-kde-probe main.cpp)
target_link_libraries(endpoint-kde-probe PRIVATE Qt6::Core Qt6::Gui Qt6::DBus KF6::IdleTime)
target_compile_options(endpoint-kde-probe PRIVATE -Wall -Wextra -Werror)
install(TARGETS endpoint-kde-probe RUNTIME DESTINATION lib/endpoint-agent)
install(FILES foreground.js DESTINATION share/endpoint-agent)
